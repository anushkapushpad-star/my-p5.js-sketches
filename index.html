let synth, drone, reverb, filter, vol; let isStarted = false; let currentBright = 100; let currentVolume = -20; let currentPitchShift = 0; function setup() { createCanvas(800, 600); angleMode(DEGREES); colorMode(HSB, 360, 100, 100, 100); } function mouseMoved() { if (!isStarted) startAudio(); updateAudio(); } function mousePressed() { if (!isStarted) startAudio(); } function updateAudio() { if (!isStarted) return; // mouseY: bottom = complete silence, top = full (0dB) let volVal = map(mouseY, height, 0, -60, 0); if (mouseY > height - 10) volVal = -Infinity; currentVolume = volVal; if (vol) vol.volume.rampTo(currentVolume, 0.3); // mouseX: left = low pitch, right = high pitch currentPitchShift = map(mouseX, 0, width, -12, 12); } async function startAudio() { if (isStarted) return; isStarted = true; await Tone.start(); reverb = new Tone.Reverb({ decay: 6, wet: 0.7 }).toDestination(); filter = new Tone.Filter(2000, 'lowpass').connect(reverb); // Master volume node — controlled by mouseY vol = new Tone.Volume(-20).connect(filter); drone = new Tone.PolySynth(Tone.Synth, { oscillator: { type: 'sine' }, envelope: { attack: 2, decay: 1, sustain: 0.8, release: 4 }, volume: -6 }).connect(vol); synth = new Tone.PolySynth(Tone.Synth, { oscillator: { type: 'triangle' }, envelope: { attack: 0.5, decay: 0.8, sustain: 0.4, release: 2 }, volume: -8 }).connect(vol); drone.triggerAttack(['C2', 'G2', 'E3']); Tone.Transport.bpm.value = 52; let step = 0; const allNotes = [ 'C2','D2','E2','G2','A2', 'C3','D3','E3','G3','A3', 'C4','D4','E4','G4','A4', 'C5','D5','E5','G5','A5' ]; Tone.Transport.scheduleRepeat((time) => { let t = constrain(map(currentBright, 30, 100, 0, 1), 0, 1); // mouseX picks center index in allNotes let centerIdx = Math.floor(map(mouseX, 0, width, 2, allNotes.length - 3)); centerIdx = constrain(centerIdx, 2, allNotes.length - 3); let offset = (step % 5) - 2; let noteIdx = constrain(centerIdx + offset, 0, allNotes.length - 1); let note = allNotes[noteIdx]; synth.triggerAttackRelease(note, '2n', time); // Filter opens/closes with mouseY filter.frequency.rampTo(map(mouseY, height, 0, 300, 5000), 0.5); // Drone shifts register with mouseX if (step % 4 === 0) { drone.releaseAll(); if (mouseX < width / 3) { drone.triggerAttack(['C2', 'G2', 'Eb3']); } else if (mouseX < (width * 2) / 3) { drone.triggerAttack(['C3', 'G3', 'E4']); } else { drone.triggerAttack(['C4', 'G4', 'E5']); } } step++; }, '2n'); Tone.Transport.start(); } function draw() { background(40, 20, 92); noStroke(); let hueShift = map(mouseX, 0, width, 0, 360); let bright = map(mouseY, 0, height, 100, 30); currentBright = bright; if (isStarted) updateAudio(); function c(h, s, b) { return color((h + hueShift) % 360, s, b * (bright / 100)); } // ---- SQUARE 1 (0,0) ---- push(); fill(c(80, 40, 75)); square(0, 0, 200); translate(100, 100); rectMode(CENTER); fill(c(50, 85, 90));
